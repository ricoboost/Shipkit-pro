## Interfaz Unificada

```typescript
// src/lib/ai/types.ts
export interface AIProvider {
  chat(messages: Message[], options?: ChatOptions): Promise<ChatResponse>;
  stream(messages: Message[], options?: ChatOptions): AsyncIterable<string>;
}
```

## Uso Básico

```typescript
import { ai } from '@/lib/ai';

// Chat simple
const response = await ai.chat([
  { role: 'user', content: 'Hello!' }
]);

console.log(response.content);
```

## Streaming

```typescript
import { ai } from '@/lib/ai';

// Respuesta en streaming
for await (const chunk of ai.stream([
  { role: 'system', content: 'You are a helpful assistant.' },
  { role: 'user', content: 'Tell me a story.' }
])) {
  process.stdout.write(chunk);
}
```

## Ruta de API

```typescript
// src/app/api/ai/chat/route.ts
import { ai } from '@/lib/ai';
import { auth } from '@/lib/auth';
import { credits } from '@/lib/payments/credits';

export async function POST(req: Request) {
  const session = await auth.getSession();
  if (!session) {
    return new Response('Unauthorized', { status: 401 });
  }

  // Verificar créditos
  const hasCredits = await credits.deduct(session.user.id, 1, 'AI chat');
  if (!hasCredits) {
    return new Response('Insufficient credits', { status: 402 });
  }

  const { messages } = await req.json();

  // Retornar respuesta en streaming
  const stream = ai.stream(messages);

  return new Response(
    new ReadableStream({
      async start(controller) {
        for await (const chunk of stream) {
          controller.enqueue(new TextEncoder().encode(chunk));
        }
        controller.close();
      },
    }),
    {
      headers: { 'Content-Type': 'text/event-stream' },
    }
  );
}
```

## Componente del Cliente

```tsx
'use client';

import { useState } from 'react';
import { useChat } from '@/hooks/use-chat';

export function ChatInterface() {
  const { messages, input, handleSubmit, isLoading } = useChat();

  return (
    <div>
      <div className="messages">
        {messages.map((m) => (
          <div key={m.id} className={m.role}>
            {m.content}
          </div>
        ))}
      </div>

      <form onSubmit={handleSubmit}>
        <input
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="Say something..."
        />
        <button disabled={isLoading}>Send</button>
      </form>
    </div>
  );
}
```

## Selección de Modelo

Cambia de modelo fácilmente:

```typescript
const response = await ai.chat(messages, {
  model: 'gpt-4-turbo',  // o 'claude-3-opus', 'llama-3-70b', etc.
  temperature: 0.7,
  maxTokens: 1000,
});
```
